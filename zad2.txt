Comm::NXTComm comm;

//PRZED UŻYCIEM SPRZETU DOKŁADNIE SPRAWDZAMY: 
//DO KTÓRYCH PORTÓW WEJŚCIOWYCH (IN) PODŁĄCZONE SĄ SENSORY,\
//DO KTÓRYCH PORTÓW WYJŚCIOWYCH (OUT) PODŁĄCZONE SĄ SERVO-MOTORY,
//STARAMY SIĘ PISAC KOD Z MOŻLIWOŚCIĄ ZATRZYMANIA ROBOTA W KAŻDEJ CHWILI,
//ROBOT POWIENIEN BYC PRZEMIESZCZONY W MIEJSCE TESTÓW ZA POŚREDNICTWEM ZDALNEGO STEROWANIA,
//CZUJNIKI SA INICJOWANE PRZED PETLA NIESKONCZONA,
//CZUJNIKI SA DEAKTYWOWANE ZA PETLA NIESKONCZONA, LUB PO KOMENDZIE k,


void foward(int speed) {
	NXT::Motor::SetForward(&comm, OUT_B, speed);
	NXT::Motor::SetForward(&comm, OUT_C, speed);
}

void backward(int speed) {
	NXT::Motor::SetReverse(&comm, OUT_B, speed);
	NXT::Motor::SetReverse(&comm, OUT_C, speed);
}

void left(int speed) {
	NXT::Motor::SetReverse(&comm, OUT_B, speed);
	NXT::Motor::SetForward(&comm, OUT_C, speed);
}

void right(int speed) {
	NXT::Motor::SetForward(&comm, OUT_B, speed);
	NXT::Motor::SetReverse(&comm, OUT_C, speed);
}

void stop() {
	NXT::Motor::SetForward(&comm, OUT_B, 0);
	NXT::Motor::SetForward(&comm, OUT_C, 0);
}

void avoid() {
	stop();

	backward(30);
	Wait(500);
	stop();

	right(30);
	Wait(300);

	stop();
}

void agressive()
{
	int light = NXT::Sensor::GetValue(&comm, IN_3);
	int distance = NXT::Sensor::GetSonarValue(&comm, IN_4);



	std::cout << "light: " << light << endl;
	std::cout << "distance: " << distance;

	std::cout << endl << endl;

	if (light > 45) {
		avoid();
	}
	else if (distance <= 40 && distance > 0) {
		foward(40);
	}
	else {
		foward(20);
		stop();
		left(20);
	}

}

void defensive()
{
	int light = NXT::Sensor::GetValue(&comm, IN_3);
	int distance = NXT::Sensor::GetSonarValue(&comm, IN_4);

	std::cout << light << endl;
	std::cout << distance;

	std::cout << endl << endl;

	if (light > 45) {
		avoid();
	}
	else if (distance <= 25 && distance > 0) {
		left(40);
		Wait(500);
		stop();

		foward(40);
		Wait(500);
		stop();

		right(50);
		Wait(500);
		stop();

		foward(40);
		Wait(500);
		stop();

	}
	else {
		foward(20);
		stop();
		left(20);
	}

}

		NXT::StartProgram(&comm, "program1");

		//zmienna używana do sterowania
		char mode = 'a';

		std::cout << "\n ___Podstawowe sterowanie do uzupelnienia___";
